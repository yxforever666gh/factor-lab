```sh
python3 - <<'PY'
import json
import sqlite3
from pathlib import Path

ROOT = Path('/home/admin/.openclaw/workspace')
CFG_DIR = ROOT / 'artifacts' / 'generated_candidate_configs'
DB = ROOT / 'artifacts' / 'factor_lab.db'

# 1) repair historical generated candidate configs
for path in CFG_DIR.glob('gen__*.json'):
    obj = json.loads(path.read_text(encoding='utf-8'))
    changed = False
    for row in obj.get('factors', []):
        name = row.get('name') or ''
        if not name.startswith('gen__'):
            continue
        if row.get('generator_operator') and row.get('left_factor_name') and row.get('right_factor_name'):
            continue
        parts = name.split('__')
        # gen__<operator>__<left>__<right>
        if len(parts) >= 4:
            row['generator_operator'] = row.get('generator_operator') or parts[1]
            row['left_factor_name'] = row.get('left_factor_name') or parts[2]
            row['right_factor_name'] = row.get('right_factor_name') or parts[3]
            changed = True
    if changed:
        path.write_text(json.dumps(obj, ensure_ascii=False, indent=2), encoding='utf-8')
        print('repaired config', path.name)

# 2) clear broken retry/pending tasks with the specific generated-operator error
conn = sqlite3.connect(DB)
conn.row_factory = sqlite3.Row
rows = list(conn.execute(\"\"\"
    select task_id, status, worker_note
    from research_tasks
    where status in ('pending','running','failed')
      and (
        last_error like '%missing base factor for generated operator%'
        or worker_note like 'retry｜自动重试 workflow%'
      )
\"\"\"))
for row in rows:
    conn.execute(
        \"update research_tasks set status='failed', finished_at_utc=datetime('now'), last_error='generated_candidate_config_repaired_reset', worker_note=coalesce(worker_note,'') || '｜auto_reset_after_generated_config_repair' where task_id=?\",
        (row['task_id'],)
    )
conn.commit()
conn.close()
print('reset tasks', len(rows))
PY
```
